O SOSLocation.ML usa PyTorch para predição de risco em zonas de desastre. Como treinamos modelos de segmentação semântica para identificar áreas críticas em imagens de satélite e dados GIS.

Modelos neurais de risco: o SOSLocation.ML

Em Março de 2026, o commit feat(ml+tests): SOSLocation.ML .NET project + 40 unit/integration tests formalizou o que já estava sendo construído: uma unidade de ML dedicada ao SOS Location. O objetivo é simples mas ambicioso — prever quais áreas serão afetadas antes que a crise se torne incontrolável.

Arquitetura do ML Unit

text 
risk-analysis-unit/
  api/            FastAPI — serve o modelo via REST e streaming gRPC
  models/         Definições PyTorch dos modelos
  training/       Scripts de treino + MLflow tracking
  data/           Pipelines de ingestão (GIS, clima, histórico)
  tests/          40 testes unitários e de integração

O ML Unit é um microsserviço Python independente que o backend .NET consulta via HTTP/gRPC quando um incidente crítico (severidade >= 7) é criado.

Modelo de segmentação semântica

O problema de identificar zonas de risco em imagens de satélite é um problema de segmentação semântica — classificar cada pixel como "risco alto", "risco médio", "seguro", "água", etc.

A arquitetura escolhida é uma U-Net modificada, clássica em problemas de segmentação de imagens médicas e geoespaciais:

python 
import torch
import torch.nn as nn
import torch.nn.functional as F

class UNetBlock(nn.Module):
    def __init__(self, in_ch: int, out_ch: int):
        super().__init__()
        self.conv = nn.Sequential(
            nn.Conv2d(in_ch, out_ch, 3, padding=1, bias=False),
            nn.BatchNorm2d(out_ch),
            nn.ReLU(inplace=True),
            nn.Conv2d(out_ch, out_ch, 3, padding=1, bias=False),
            nn.BatchNorm2d(out_ch),
            nn.ReLU(inplace=True),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.conv(x)


class SOSRiskSegmenter(nn.Module):
    """U-Net para segmentação de risco em imagens multiespectral + DEM."""

    RISK_CLASSES = ['safe', 'low_risk', 'medium_risk', 'high_risk', 'critical', 'water']

    def __init__(self, in_channels: int = 6, num_classes: int = 6):
        # in_channels: RGB (3) + DEM (1) + NDVI (1) + moisture index (1)
        super().__init__()
        # Encoder
        self.enc1 = UNetBlock(in_channels, 64)
        self.enc2 = UNetBlock(64, 128)
        self.enc3 = UNetBlock(128, 256)
        self.enc4 = UNetBlock(256, 512)
        # Bottleneck
        self.bottleneck = UNetBlock(512, 1024)
        # Decoder
        self.up4 = nn.ConvTranspose2d(1024, 512, 2, stride=2)
        self.dec4 = UNetBlock(1024, 512)
        self.up3 = nn.ConvTranspose2d(512, 256, 2, stride=2)
        self.dec3 = UNetBlock(512, 256)
        self.up2 = nn.ConvTranspose2d(256, 128, 2, stride=2)
        self.dec2 = UNetBlock(256, 128)
        self.up1 = nn.ConvTranspose2d(128, 64, 2, stride=2)
        self.dec1 = UNetBlock(128, 64)
        # Output
        self.out = nn.Conv2d(64, num_classes, 1)
        self.pool = nn.MaxPool2d(2)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        e1 = self.enc1(x)
        e2 = self.enc2(self.pool(e1))
        e3 = self.enc3(self.pool(e2))
        e4 = self.enc4(self.pool(e3))
        b  = self.bottleneck(self.pool(e4))
        d4 = self.dec4(torch.cat([self.up4(b),  e4], dim=1))
        d3 = self.dec3(torch.cat([self.up3(d4), e3], dim=1))
        d2 = self.dec2(torch.cat([self.up2(d3), e2], dim=1))
        d1 = self.dec1(torch.cat([self.up1(d2), e1], dim=1))
        return self.out(d1)  # (B, num_classes, H, W)

Features de entrada

O modelo recebe 6 canais por tile (256x256 pixels, ~100m/pixel):

CanalFonteDescrição
R, G, BSentinel-2 / LandsatImagem óptica
DEMSRTMGL1Elevação em metros
NDVICalculadoÍndice de vegetação
SMICalculadoSoil Moisture Index
python 
def prepare_tile(
    rgb: np.ndarray,        # (3, 256, 256) float32 [0,1]
    dem: np.ndarray,        # (256, 256) float32 metros
    ndvi: np.ndarray,       # (256, 256) float32 [-1, 1]
    smi: np.ndarray,        # (256, 256) float32 [0, 1]
) -> torch.Tensor:
    dem_norm  = (dem - dem.mean()) / (dem.std() + 1e-6)
    x = np.stack([rgb[0], rgb[1], rgb[2], dem_norm, ndvi, smi], axis=0)
    return torch.from_numpy(x).unsqueeze(0)  # (1, 6, 256, 256)

python 
class RiskDataset(Dataset):
    def __init__(self, csv_path: str, tiles_dir: str, augment: bool = True):
        self.df = pd.read_csv(csv_path)
        self.tiles_dir = Path(tiles_dir)
        self.augment = augment

    def __getitem__(self, idx: int):
        row   = self.df.iloc[idx]
        tile  = np.load(self.tiles_dir / row['tile_file'])  # (6, 256, 256)
        mask  = np.load(self.tiles_dir / row['mask_file'])  # (256, 256) int labels

        if self.augment:
            tile, mask = self._random_flip(tile, mask)

        return torch.from_numpy(tile), torch.from_numpy(mask).long()

Servindo o modelo via FastAPI

python 
# api/main.py
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
import torch, io, json

app = FastAPI(title="SOSLocation Risk Model API")
model = SOSRiskSegmenter().eval()
model.load_state_dict(torch.load("weights/best_model.pt", map_location="cpu"))

@app.post("/risk/predict")
async def predict_risk(payload: RiskRequest):
    tile = prepare_tile(
        rgb=np.array(payload.rgb), dem=np.array(payload.dem),
        ndvi=np.array(payload.ndvi), smi=np.array(payload.smi)
    )
    with torch.no_grad():
        logits = model(tile)                  # (1, 6, 256, 256)
        probs  = torch.softmax(logits, dim=1) # probabilidades por classe
        pred   = logits.argmax(dim=1)         # classe com maior probabilidade

    return {
        "prediction":    pred[0].numpy().tolist(),
        "probabilities": probs[0].numpy().tolist(),
        "risk_level":    int(pred.max().item()),
        "critical_pct":  float((pred == 4).float().mean().item()) * 100,
    }
